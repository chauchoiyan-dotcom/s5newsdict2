<script>
    const passages = [
        "The charity has since evolved into an environmental NGO that not only engages retirees, such as Lai, in urban farming but also educates the public on eco-friendly ways to grow and consume food, including planting and buying in season to reduce the need for fertilisers and pesticides.",
        "The charity is one of 13 NGOs supported by OSC this year. The funding will allow its Golden Greens Programme to promote gardening and nature-based activities that empower 100 low-income, socially isolated elderly residents in Kwai Tsing and Sham Shui Po next year, whether in elderly homes or public housing estates."
    ];

    let currentIdx = 0;
    let lastScore = 0;

    function switchSentence(index) {
        currentIdx = index;
        document.getElementById('display-idx').innerText = (index + 1) + " / 2";
        document.querySelectorAll('.tab-btn').forEach((btn, i) => btn.classList.toggle('active', i === index));
        clearAll();
    }

    function speakText() {
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(passages[currentIdx]);
        utterance.lang = 'en-GB';
        utterance.rate = 0.7;
        window.speechSynthesis.speak(utterance);
    }

    function showHint() {
        const words = passages[currentIdx].split(' ');
        const hints = words.map(w => w[0] + w.slice(1).replace(/[a-zA-Z]/g, "_"));
        const display = document.getElementById('hint-display');
        display.innerText = hints.join(' ');
        display.style.display = 'block';
    }

    function clearAll() {
        document.getElementById('userInput').value = "";
        document.getElementById('feedback').style.display = 'none';
        document.getElementById('hint-display').style.display = 'none';
        document.getElementById('display-score').innerText = "0%";
    }

    function checkAnswer() {
        const userText = document.getElementById('userInput').value.trim();
        const feedbackDiv = document.getElementById('feedback');
        if (!userText) return;

        const targetWords = passages[currentIdx].split(' ');
        const userWords = userText.split(/\s+/);
        let correctCount = 0;

        let resultHTML = "<strong>Correction:</strong><br>";
        targetWords.forEach((word, index) => {
            const cleanTarget = word.replace(/[.,\/#!$%\^&\*;:{}=\-_`~()]/g,"").toLowerCase();
            const cleanUser = userWords[index] ? userWords[index].replace(/[.,\/#!$%\^&\*;:{}=\-_`~()]/g,"").toLowerCase() : "";
            
            if (cleanTarget === cleanUser) {
                resultHTML += `<span class="correct">${word}</span> `;
                correctCount++;
            } else {
                resultHTML += `<span class="incorrect">${word}</span> `;
            }
        });

        lastScore = Math.round((correctCount / targetWords.length) * 100);
        document.getElementById('display-score').innerText = lastScore + "%";
        feedbackDiv.innerHTML = resultHTML + `<br><br><strong>Score: ${correctCount}/${targetWords.length} words correct.</strong>`;
        feedbackDiv.style.display = 'block';
    }

    function downloadReport() {
        const name = document.getElementById('studentName').value || "Student";
        const content = `Dictation Revision Report\nName: ${name}\nDate: ${new Date().toLocaleDateString()}\nSection: ${currentIdx + 1}\nScore: ${lastScore}%`;
        const blob = new Blob([content], { type: 'text/plain' });
        const a = document.createElement('a');
        a.href = URL.createObjectURL(blob);
        a.download = `${name}_Dictation_Score.txt`;
        a.click();
    }
</script>
